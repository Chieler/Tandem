# cntctr 0.2.0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Gathers that survive long waits and never lose replies, visible agent state with usage-limit detection and fallback, board posts that do not interrupt, and the smaller fixes from the 2026-11-07 orchestrator feedback.

**Architecture:** Everything stays in `cntctr.c` (plus one board query in `kb.c`), following the existing single-file pattern. A screen classifier (`pane_state`) feeds `list`, gather and a new per-process watcher thread. Gather, blocking `send` and blocking `spawn` share one wait loop (`wait_tasks`). New per-task and per-pane state lives in tmux options, like the existing `@cntctr_from_/to_/wait_` options.

**Tech Stack:** C11, json-c, SQLite, POSIX regex, tmux 3.x. Tests: `test.sh` (bash, jq, private tmux server).

**Spec:** `docs/superpowers/specs/2026-11-08-qol-design.md`

## Global Constraints

- Branch `feat/qol`, based on `main` (`v0.1.0`). Never commit to `feat/fleet-tools` or `main`.
- Server version string is `"0.2.0"` (already set in `initialize`).
- No new dependencies. Build with `make cntctr`; `make test` (ASan + UBSan build) must pass after every task.
- Nothing cntctr sends for its own bookkeeping (heartbeats) is ever typed into a pane.
- The watcher never types into a pane whose state is `busy` or `permission`.
- Every new tool argument appears in `TOOLS` (tools/list) with a description.
- Commit messages: Conventional Commits, ending with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.
- Run tests with `make test`; a single section can be tried by running `make cntctr-test && BIN=./cntctr-test bash test.sh 2>&1 | grep -E 'FAIL|passed'`.

## Review Focus

1. Client aborts a blocking gather without sending `notifications/cancelled` (old Claude Code, crash): replies already claimed are in a result nobody reads. Expected: the 3 h client timeout plus heartbeats make this rare; no extra handling. Reviewer checks heartbeats actually flow (Task 3 test).
2. A reply arrives in the same instant the gather times out or is cancelled: it must land exactly once, in the result or in the pane. Covered by the claim order in `finish_pending` (Task 4) and the existing `delete-buffer` claim in `tool_reply`.
3. A worker pane whose scrollback still shows an old limit message after the reset: it reads `limited` until that text scrolls out of the last 15 lines. Expected: acceptable; Claude's auto-continue prints new output that pushes it out. Task 1 test pins that only the last 15 lines count.
4. Two orchestrators each run a watcher over the same worker: a limit notice or board batch must be delivered once. Covered by the tmux-option dedup in Tasks 7 and 8 (set before inject; unset on inject failure).
5. Gather result text that is JSON (`reply_format`) must not get the unread-board line appended. Task 8 test pins this.

  - `/* ---- watcher ---- */` before `dispatch`: served panes, `watch_round`, `watcher`.
- `kb.c` / `kb.h`: `kb_unread`.

- `README.md`, `docs/BACKLOG.md`: Task 13.

```c
/* Last 15 lines of `pane` (caller frees), or NULL if the pane is gone. */
static char *pane_tail(const char *pane) {
    char *screen;
    if (TMUX(NULL, &screen, "capture-pane", "-p", "-t", pane) != 0) {
        free(screen);
        return NULL;
    }
    char *tail = screen + strlen(screen);   /* run() already dropped trailing blank lines */
    for (int lines = 0; tail > screen; tail--)
        if (tail[-1] == '\n' && ++lines == 15) break;
    memmove(screen, tail, strlen(tail) + 1);
    return screen;
}

static int match_any(const regex_t *re, size_t n, const char *s) {
    for (size_t i = 0; i < n; i++)
        if (regexec(&re[i], s, 0, NULL, 0) == 0) return 1;
    return 0;
}

static int awaiting_permission(const char *pane) {
    char *tail = pane_tail(pane);   /* a missing pane fails later, in paste-buffer */
    int hit = tail && match_any(prompt_re, sizeof PROMPTS / sizeof *PROMPTS, tail);
    free(tail);
    return hit;
}

/* What `pane` is doing, from its screen. Order matters: a limit screen can also show
 * "esc to cancel", and permission dialogs can show "Esc to cancel". */
static enum pstate pane_state(const char *pane, char **resets) {
    if (resets) *resets = NULL;
    if (!pane_alive(pane)) return P_DEAD;
    char *tail = pane_tail(pane);
    if (!tail) return P_DEAD;
    enum pstate s = P_IDLE;
    regmatch_t m[4];
    if (match_any(limit_re, sizeof LIMITS / sizeof *LIMITS, tail)) {
        s = P_LIMITED;
        if (resets && regexec(&resets_re, tail, 4, m, 0) == 0 && m[3].rm_so >= 0) {
            int n = (int)(m[3].rm_eo - m[3].rm_so);
            while (n > 0 && tail[m[3].rm_so + n - 1] == ' ') n--;
            *resets = fmt("%.*s", n, tail + m[3].rm_so);
        }
    } else if (match_any(prompt_re, sizeof PROMPTS / sizeof *PROMPTS, tail)) {
        s = P_PERMISSION;
    } else if (match_any(busy_re, sizeof BUSY / sizeof *BUSY, tail)) {
        s = P_BUSY;
    }
    free(tail);
    return s;
}

/* All server options, one "name value" per line (caller frees). */
static char *server_opts(void) {
    char *out;
    if (TMUX(NULL, &out, "show", "-s") == 0) return out;
    free(out);
    return strdup("");
}

/* Value of option `name` in `opts` from server_opts() (caller frees), or NULL. */
static char *opt_of(const char *opts, const char *name) {
    size_t n = strlen(name);
    for (const char *l = opts; *l; l += strcspn(l, "\n") + (l[strcspn(l, "\n")] == '\n')) {
        if (strncmp(l, name, n) == 0 && l[n] == ' ') {
            const char *v = l + n + 1;
            size_t len = strcspn(v, "\n");
            if (len >= 2 && *v == '"' && v[len - 1] == '"') v++, len -= 2;   /* tmux quotes values with spaces */
            return fmt("%.*s", (int)len, v);
        }
    }
    return NULL;
}
```

```c
/* ---- waiting ---- */

struct gathered {
    const char *kind, *agent, *status;
    char id[9], *pane, *text, *resets;
    json_object *task;
    char fb_agent[128], fb_id[9];
};

static void unwait(const char *id) {
    char opt[48];
    snprintf(opt, sizeof opt, "@cntctr_wait_%s", id);
    free(TMUXE(NULL, "set", "-su", opt));
}

/* One check of a pending task. Claims the reply if one arrived. 1 when it->status got set. */
static int poll_task(struct gathered *it) {
    char buf[40];
    snprintf(buf, sizeof buf, "cntctr-reply-%s", it->id);
    if (buffer_exists(buf)) {
        unwait(it->id);   /* stop repliers from signalling us, then claim */
        if ((it->text = buffer_take(buf))) {
            untrack(it->id);
            it->status = "answered";
            return 1;
        }
    }
    enum pstate st = pane_state(it->pane, &it->resets);
    if (st == P_DEAD) {
        unwait(it->id);
        if ((it->text = buffer_take(buf))) {   /* answered just before exiting */
            untrack(it->id);
            it->status = "answered";
            return 1;
        }
        untrack(it->id);
        it->text = fmt("agent %s exited before replying (id=%s)", it->agent, it->id);
        it->status = "dead";
        return 1;
    }
    if (st == P_LIMITED) {
        unwait(it->id);   /* a reply after the reset reaches the caller's pane */
        it->status = "limited";
        return 1;
    }
    free(it->resets);
    it->resets = NULL;
    return 0;
}

#define POLL_MS 1000   /* ponytail: polls every pending task each second; tmux wait-for per task if gathers grow large */

/* Wait until every task has a status, the deadline passes, or the call is cancelled. */
static void wait_tasks(struct gathered *items, size_t n, time_t deadline) {
    for (;;) {
        size_t pending = 0;
        for (size_t i = 0; i < n; i++)
            if (!items[i].status && !poll_task(&items[i])) pending++;
        if (!pending || cancelled() || time(NULL) >= deadline) return;
        struct timespec pause = {POLL_MS / 1000, (POLL_MS % 1000) * 1000000L};
        nanosleep(&pause, NULL);
    }
}

/* Settle tasks still pending after wait_tasks. */
static void finish_pending(struct gathered *items, size_t n) {
    for (size_t i = 0; i < n; i++)
        if (!items[i].status) {
            unwait(items[i].id);
            items[i].status = "timed_out";
        }
}

static char *gather_json(struct gathered *items, size_t n) {
    json_object *out = json_object_new_object(), *results = json_object_new_array();
    int complete = 1;
    for (size_t i = 0; i < n; i++) {
        struct gathered *it = &items[i];
        int answered = strcmp(it->status, "answered") == 0;
        if (!answered) complete = 0;
        json_object *e = json_object_new_object();
        json_object_object_add(e, "kind", json_object_new_string(it->kind ? it->kind : ""));
        json_object_object_add(e, "agent", json_object_new_string(it->agent ? it->agent : ""));
        json_object_object_add(e, "status", json_object_new_string(it->status));
        if (*it->id) json_object_object_add(e, "id", json_object_new_string(it->id));
        if (it->resets) json_object_object_add(e, "resets", json_object_new_string(it->resets));
        if (it->text) json_object_object_add(e, answered ? "text" : "error", json_object_new_string(it->text));
        json_object_array_add(results, e);
    }
    json_object_object_add(out, "complete", json_object_new_boolean(complete));
    json_object_object_add(out, "results", results);
    char *wire = strdup(json_object_to_json_string_ext(out, JSON_C_TO_STRING_PLAIN));
    json_object_put(out);
    return wire;
}

static void free_items(struct gathered *items, size_t n) {
    for (size_t i = 0; i < n; i++) {
        free(items[i].pane);
        free(items[i].text);
        free(items[i].resets);
    }
    free(items);
}

/* Blocking send or spawn: one task through the same loop as gather. */
static int wait_one(const char *id, char *pane, const char *who, json_object *a, char **res) {
    struct gathered it = {.kind = "send", .agent = who, .pane = pane, .task = a};
    snprintf(it.id, sizeof it.id, "%s", id);
    wait_tasks(&it, 1, time(NULL) + targ(a));
    finish_pending(&it, 1);
    int r;
    if (strcmp(it.status, "answered") == 0) r = ok(res, it.text), it.text = NULL;
    else if (strcmp(it.status, "dead") == 0) r = fail(res, it.text), it.text = NULL;
    else if (strcmp(it.status, "limited") == 0)
        r = ok(res, fmt("agent %s hit its usage limit%s%s%s; reply will arrive in your pane (id=%s)", who,
                        it.resets ? " (resets " : "", it.resets ? it.resets : "", it.resets ? ")" : "", id));
    else r = ok(res, fmt("timed out; reply will arrive in your pane (id=%s)", id));
    free(it.text);
    free(it.resets);
    return r;
}
```

```c
/* ---- watcher ---- */

static int watch_secs = 30;   /* CNTCTR_WATCH_SECS */
static pthread_mutex_t served_mu = PTHREAD_MUTEX_INITIALIZER;
static char served[64][32];   /* panes this process has handled calls for; one unless Codex shares us */
static int nserved;
static pthread_once_t watcher_once = PTHREAD_ONCE_INIT;

static int is_served(const char *pane) {
    pthread_mutex_lock(&served_mu);
    int hit = 0;
    for (int i = 0; i < nserved && !hit; i++) hit = strcmp(served[i], pane) == 0;
    pthread_mutex_unlock(&served_mu);
    return hit;
}

/* Type a notice into `pane` only while it is idle. 0 if delivered. */
static int notify_idle(const char *pane, const char *text) {
    if (pane_state(pane, NULL) != P_IDLE) return -1;
    char *err = inject(pane, "watcher", text);
    int r = err ? -1 : 0;
    free(err);
    return r;
}

static void watch_task(const char *opts, const char *id, const char *sender) {
    char opt[48], *v;
    snprintf(opt, sizeof opt, "@cntctr_wait_%s", id);
    if ((v = opt_of(opts, opt))) { free(v); return; }   /* a blocking call watches this one itself */
    snprintf(opt, sizeof opt, "@cntctr_to_%s", id);
    char *target = opt_of(opts, opt);
    if (!target) return;
    char *who = pane_name(target), *resets = NULL;
    snprintf(opt, sizeof opt, "@cntctr_limited_%s", id);
    char *seen = opt_of(opts, opt);
    if (!seen && pane_state(target, &resets) == P_LIMITED) {
        char nid[9], *npane = NULL, agent[128] = "", *err = reroute(id, who ? who : "?", target, 0, nid, &npane, agent);
        char *moved = err ? (strcmp(err, "no fallback") ? fmt("; fallback failed: %s", err) : strdup(""))
                          : fmt("; re-sent to %s as %s", agent, nid);
        char *note = fmt("[cntctr] %s hit usage limit%s%s%s; task %s not done%s", who ? who : "?",
                         resets ? " (resets " : "", resets ? resets : "", resets ? ")" : "", id, moved);
        free(TMUXE(NULL, "set", "-s", opt, "1"));   /* claim before typing, so a second watcher skips it */
        if (notify_idle(sender, note) != 0) free(TMUXE(NULL, "set", "-su", opt));
        free(note);
        free(moved);
        free(err);
        free(npane);
    }
    snprintf(opt, sizeof opt, "@cntctr_done_%s", id);
    char *path = opt_of(opts, opt);
    if (path && access(path, F_OK) == 0) {
        char *note = fmt("[cntctr] %s task %s done: wrote %s", who ? who : "?", id, path);
        free(TMUXE(NULL, "set", "-su", opt));
        if (notify_idle(sender, note) != 0) free(TMUXE(NULL, "set", "-s", opt, path));
        free(note);
    }
    free(path);
    free(seen);
    free(resets);
    free(who);
    free(target);
}
```

```c
static void watch_round(void) {
    char *opts = server_opts();
    for (const char *l = opts; *l; l += strcspn(l, "\n") + (l[strcspn(l, "\n")] == '\n')) {
        char id[9], sender[32];
        if (sscanf(l, "@cntctr_from_%8[0-9a-f] %31s", id, sender) == 2 && is_served(sender))
            watch_task(opts, id, sender);
    }
    free(opts);
}

static void *watcher(void *arg) {
    (void)arg;
    for (;;) {
        sleep((unsigned)watch_secs);
        watch_round();
    }
    return NULL;
}

static void start_watcher(void) {
    pthread_t t;
    if (pthread_create(&t, NULL, watcher, NULL) == 0) pthread_detach(t);
}

static void serve(const char *pane) {
    if (!*pane) return;
    pthread_mutex_lock(&served_mu);
    int known = 0;
    for (int i = 0; i < nserved && !known; i++) known = strcmp(served[i], pane) == 0;
    if (!known && nserved < 64) snprintf(served[nserved++], sizeof served[0], "%s", pane);
    pthread_mutex_unlock(&served_mu);
    pthread_once(&watcher_once, start_watcher);
}
```

```c
char *kb_unread(sqlite3 *db, const char *scope, const char *me, long long after, long long *newest, int *count) {
    sqlite3_stmt *st = prep(db, "SELECT rowid, sender, text FROM posts WHERE scope = ?1 AND rowid > ?2 ORDER BY rowid");
    sqlite3_bind_text(st, 1, scope, -1, SQLITE_TRANSIENT);
    sqlite3_bind_int64(st, 2, after);
    char *out = strdup("");
    *newest = after;
    *count = 0;
    while (sqlite3_step(st) == SQLITE_ROW) {
        *newest = sqlite3_column_int64(st, 0);
        if (strcmp(col(st, 1), me) == 0) continue;   /* our own posts count as seen */
        char *t;
        if (asprintf(&t, "%s- %s: %s\n", out, col(st, 1), col(st, 2)) < 0) break;
        free(out);
        out = t;
        (*count)++;
    }
    sqlite3_finalize(st);
    return out;
}
```

```c
static char *unread_for(const char *pane, const char *name, long long *newest, int *count) {
    char *err = NULL;

    free(err);

    char *out = kb_unread(db, p.key, name, seen_of(name), newest, count);
```

```c
/* Fail with "<tool> needs 'a', 'b' (missing: b). Example: <tool> <example>" if a field is absent or empty. */
static int need(char **res, json_object *a, const char *tool, const char *const *fields, const char *example) {
    char *all = strdup(""), *missing = strdup(""), *t;
    for (int i = 0; fields[i]; i++) {
        const char *v = sarg(a, fields[i]);
        t = fmt("%s%s'%s'", all, i ? ", " : "", fields[i]);
        free(all);
        all = t;
        if (!v || !*v) {
            t = fmt("%s%s%s", missing, *missing ? ", " : "", fields[i]);
            free(missing);
            missing = t;
        }
    }
    int bad = *missing != '\0';
    if (bad) fail(res, fmt("%s needs %s (missing: %s). Example: %s %s", tool, all, missing, tool, example));
    free(all);
    free(missing);
    return bad;
}
```
